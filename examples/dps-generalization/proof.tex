\section*{Abstract}

This document presents a formal proof that characterizes elements of the minimal tensor product of proper cones in terms of symmetric extensions. Specifically, for proper cones $\C_A$ and $\C_B$, an element $x \in \C_A \tmax \C_B$ belongs to the minimal tensor product $\C_A \tmin \C_B$ if and only if $x$ admits symmetric extensions to $k$ copies for all $k \geq 1$. This generalizes the Doherty--Parrilo--Spedalieri (DPS) characterization of separable quantum states to arbitrary proper convex cones.

\section{Definitions and Assumptions}

\begin{definition}[Proper Cone]
A convex cone $\C \subset V$ in a finite-dimensional real vector space $V$ is \emph{proper} if it is:
\begin{enumerate}
\item \emph{Closed}: $\C = \overline{\C}$
\item \emph{Pointed}: $\C \cap (-\C) = \{0\}$
\item \emph{Generating}: $\C - \C = V$
\end{enumerate}
\end{definition}

\begin{definition}[Dual Cone]
Given a proper cone $\C \subset V$, its dual cone is
\[
\C^* = \{f \in V^* : f(x) \geq 0 \text{ for all } x \in \C\}.
\]
By the bipolar theorem, $(\C^*)^* = \C$.
\end{definition}

\begin{definition}[Minimal Tensor Product] \label{def:min}
The minimal tensor product of proper cones $\C_A \subset V_A$ and $\C_B \subset V_B$ is
\[
\C_A \tmin \C_B = \conv\{a \otimes b : a \in \C_A, b \in \C_B\} \subset V_A \otimes V_B.
\]
\end{definition}

\begin{definition}[Maximal Tensor Product] \label{def:max}
The maximal tensor product is the dual of the minimal tensor product of duals:
\[
\C_A \tmax \C_B = (\C_A^* \tmin \C_B^*)^* = \{z \in V_A \otimes V_B : (f \otimes g)(z) \geq 0 \text{ for all } f \in \C_A^*, g \in \C_B^*\}.
\]
\end{definition}

\begin{definition}[Symmetric Extension] \label{def:symext}
Fix $\phi \in \inte(\C_A^*)$. An element $x \in \C_A \tmax \C_B$ has a \emph{symmetric extension to $k$ copies} if there exists $y_k \in \C_A^{\tmax k} \tmax \C_B$ such that:
\begin{enumerate}
\item $y_k$ is invariant under permutations of the $\C_A$ factors
\item $(\phi^{\otimes(k-1)} \otimes \mathrm{id}_A \otimes \mathrm{id}_B)(y_k) = x$
\end{enumerate}
\end{definition}

\begin{axiom}[de Finetti Theorem] \label{ax:definetti}
For an exchangeable sequence of random variables on a compact convex set in finite dimensions, there exists a unique probability measure such that the sequence has an i.i.d.\ representation under that measure.
\end{axiom}

\section{Main Result}

\begin{theorem}[Symmetric Extension Characterization] \label{thm:main}
Let $\C_A$ and $\C_B$ be proper cones, and let $\phi \in \inte(\C_A^*)$. For $x \in \C_A \tmax \C_B$:
\[
x \text{ has symmetric extensions to } k \text{ copies for all } k \geq 1 \iff x \in \C_A \tmin \C_B.
\]
\end{theorem}

\begin{proof}
The proof proceeds in two directions.

\subsection*{Direction ($\Leftarrow$): Minimal tensor implies symmetric extensions}

\begin{enumerate}[leftmargin=2em]
\proofstep{1}{1} Assume $x \in \C_A \tmin \C_B$. \textit{(Local assumption :2-fwd001)}

\proofstep{1}{2} By Definition~\ref{def:min}, $x = \sum_{i=1}^n \lambda_i (a_i \otimes b_i)$ where $\lambda_i \geq 0$, $a_i \in \C_A$, $b_i \in \C_B$. \textit{(Definition expansion :2-fwd002)}

\proofstep{1}{3} For any $k \geq 1$, define $y_k = \sum_{i=1}^n \lambda_i (a_i^{\otimes k} \otimes b_i)$. \textit{(Construction :2-fwd003)}

\proofstep{1}{4} $y_k \in \C_A^{\tmax k} \tmax \C_B$ because $a_i^{\otimes k} \otimes b_i \in \C_A^{\tmin k} \tmin \C_B \subseteq \C_A^{\tmax k} \tmax \C_B$. \textit{(Substitution :2-fwd004)}

\proofstep{1}{5} $y_k$ is symmetric under permutations: for any $\pi \in S_k$, $\pi \cdot (a_i^{\otimes k}) = a_i^{\otimes k}$ since $a_i^{\otimes k}$ is the $k$-fold tensor product of identical elements. \textit{(Algebraic :2-fwd005)}

\proofstep{1}{6} Compute $(\phi^{\otimes(k-1)} \otimes \mathrm{id}_A \otimes \mathrm{id}_B)(y_k) = \sum_i \lambda_i \phi(a_i)^{k-1} (a_i \otimes b_i)$. \textit{(Algebraic :2-fwd006)}

\proofstep{1}{7} WLOG, assume $\phi(a_i) = 1$ for all $i$ by rescaling $a_i \mapsto a_i/\phi(a_i)$ and $\lambda_i \mapsto \lambda_i \phi(a_i)$, using $\phi(a_i) > 0$ since $\phi \in \inte(\C_A^*)$. \textit{(Normalization :2-fwd007)}

\proofstep{1}{8} With this normalization: $(\phi^{\otimes(k-1)} \otimes \mathrm{id}_A \otimes \mathrm{id}_B)(y_k) = \sum_i \lambda_i (a_i \otimes b_i) = x$. \textit{(Substitution :2-fwd008)}

\proofstep{1}{9} Thus $y_k$ satisfies all requirements of Definition~\ref{def:symext}. \textit{(Conjunction :2-fwd009)}

\proofstep{1}{10} \textbf{Direction ($\Leftarrow$) complete.} \textit{(Discharge :2-fwd010)}
\end{enumerate}

\subsection*{Direction ($\Rightarrow$): Symmetric extensions imply minimal tensor}

\begin{enumerate}[leftmargin=2em]
\proofstep{1}{1} Assume $x$ has symmetric extensions $y_k$ to $k$ copies for all $k \geq 1$. \textit{(Local assumption :2-bwd001)}

\proofstep{1}{2} Since $\C_B$ is generating ($\C_B - \C_B = V_B$), there exists a basis $\{e_j\}_{j=1}^{\dim V_B}$ of $V_B$ with each $e_j \in \inte(\C_B)$. \textit{(Existence :2-bwd002)}

\proofstep{1}{3} Each $e_j$ defines a dual functional $e_j^* \in V_B^*$. For any $g \in \C_B^*$, $g(e_j) > 0$ since $e_j \in \inte(\C_B)$. \textit{(Definition :2-bwd003)}

\proofstep{1}{4} For each $j$ and $k$, define $\bar{x}_{e_j,k} = (\phi^{\otimes(k-1)} \otimes \mathrm{id}_A \otimes e_j^*)(y_k) \in V_A^{\otimes k}$. \textit{(Construction :2-bwd004)}

\proofstep{1}{5} $\bar{x}_{e_j,k} \in \C_A^{\tmax k}$ because $y_k \in \C_A^{\tmax k} \tmax \C_B$ and $e_j \in \C_B$ implies $(\mathrm{id} \otimes e_j^*)$ preserves cone positivity. \textit{(Modus ponens :2-bwd005)}

\proofstep{1}{6} $\bar{x}_{e_j,k}$ is symmetric under permutations because $y_k$ is symmetric. \textit{(Algebraic :2-bwd006)}

\proofstep{1}{7} The sequence $\{\bar{x}_{e_j,k}\}_{k \geq 1}$ is exchangeable: for any $l > 0$, $\bar{x}_{e_j,k} = (\phi^{\otimes l} \otimes \mathrm{id}_A^{\otimes k})(\bar{x}_{e_j,k+l})$. \textit{(Algebraic :2-bwd007)}

\proofstep{1}{8} By Axiom~\ref{ax:definetti} (de Finetti), there exists a unique probability measure $\mu_{e_j}$ on the normalized base $B_A = \{a \in \C_A : \phi(a) = 1\}$ such that $\bar{x}_{e_j,k} = \int_{B_A} a^{\otimes k} \, d\mu_{e_j}(a)$. \textit{(External :2-bwd008)}

\proofstep{1}{9} For each $a \in B_A$, define $F_a: V_B \to \mathbb{R}$ by $F_a(e_j) = d\mu_{e_j}(a)$. This is well-defined by uniqueness of the de Finetti measure. \textit{(Construction :2-bwd009)}

\proofstep{1}{10} $F_a$ extends linearly to $V_B$ and is positive on convex combinations of basis elements $\{e_j\}$, hence on $\C_B$. \textit{(Algebraic :2-bwd010)}

\proofstep{1}{11} Since $F_a \in \C_B^*$ is positive linear on $V_B$, by the bipolar theorem $(\C_B^*)^* = \C_B$, there exists unique $\sigma_a \in \C_B$ such that $F_a(b) = \langle \sigma_a, b \rangle$ for all $b \in V_B$. \textit{(Riesz :2-bwd011)}

\proofstep{1}{12} Define $P(a) = \psi(\sigma_a)$ where $\psi \in \inte(\C_B^*)$ is a normalization functional. Then $P$ is a probability density on $B_A$. \textit{(Construction :2-bwd012)}

\proofstep{1}{13} The symmetric extensions satisfy: $y_k = \int_{B_A} a^{\otimes k} \otimes \sigma_a \, dP(a)$. \textit{(Algebraic :2-bwd013)}

\proofstep{1}{14} Taking the marginal: $x = \int_{B_A} a \otimes \sigma_a \, dP(a)$. \textit{(Algebraic :2-bwd014)}

\proofstep{1}{15} The integral is a convex combination of elements $a \otimes \sigma_a$ where $a \in \C_A$ and $\sigma_a \in \C_B$. \textit{(Definition :2-bwd015)}

\proofstep{1}{16} By Definition~\ref{def:min}, $x \in \C_A \tmin \C_B$. \textit{(Definition :2-bwd016)}

\proofstep{1}{17} \textbf{Direction ($\Rightarrow$) complete.} \textit{(Discharge :2-bwd017)}
\end{enumerate}

\subsection*{Conclusion}

By conjunction of ($\Leftarrow$) and ($\Rightarrow$), the theorem is proved. \textbf{QED.}

\end{proof}

structure ProperCone (V : Type*) [AddCommGroup V] [Module R V]
    [TopologicalSpace V] where
  carrier : Set V
  nonempty : carrier.Nonempty
  closed : IsClosed carrier
  cone : forall x mem carrier, forall r : R, 0 <= r -> r . x mem carrier
  convex : Convex R carrier
  pointed : forall x, x mem carrier -> -x mem carrier -> x = 0
  generating : forall v : V, exists c1 c2, c1 mem carrier /\
               c2 mem carrier /\ v = c1 - c2

/-- Minimal tensor product: conv{a (x) b : a in C_A, b in C_B} -/
def minTensorProduct (CA : ProperCone VA) (CB : ProperCone VB) :
    Set (VA (x)[R] VB) :=
  convexHull R {z | exists a mem CA.carrier, exists b mem CB.carrier,
                   z = a (x)t[R] b}

/-- Maximal tensor product: (C_A* (x)_min C_B*)* -/
def maxTensorProduct (CA : ProperCone VA) (CB : ProperCone VB) :
    Set (VA (x)[R] VB) :=
  {z | forall f mem CA.dual, forall g mem CB.dual,
       0 <= TensorProduct.lift (f.smulRight g) z}
